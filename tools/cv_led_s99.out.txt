sh-5.2# cat /etc/init.d/S99camevision 2>/dev/null
#!/bin/sh
# Boot helpers. Never change the USB gadget. Never insmod rockit.
# Wi-Fi always joins /userdata/wpa_camevision.conf (Camemake R&D center).
chmod 755 /userdata/camevision-wifi.sh /userdata/camevision-imu.sh /userdata/camevision-stream.sh 2>/dev/null
( sleep 8
  if [ -x /userdata/camevision-imu.sh ]; then
    /userdata/camevision-imu.sh >/userdata/imu-boot.log 2>&1
  fi
) &
( sleep 6
  i=0
  while [ $i -lt 4 ]; do
    /userdata/camevision-wifi.sh >/userdata/wifi-boot.log 2>&1
    if ip -4 addr show wlan0 2>/dev/null | grep -q 'inet '; then
      break
    fi
    i=$((i + 1))
    sleep 8
  done
) &
( sleep 12
  if [ -e /userdata/uvc-webcam.on ]; then
    echo "skip RTSP grabber; UVC webcam owns ISP" >/userdata/boot-stream.log
  elif [ -e /dev/video13 ]; then
    /userdata/camevision-stream.sh >/userdata/boot-stream.log 2>&1
  fi
) &
exit 0
